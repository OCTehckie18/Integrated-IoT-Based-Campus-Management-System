float readLux() {
  int raw = analogRead(PIN_LDR);
  float v = raw / 4095.0 * 3.3;
  if (raw <= 0) return 100000;            // very bright
  if (v >= 3.29) return 0.1;              // very dark
  float r = 2000 * v / (1 - v / 3.3);     // Wokwi LDR model (GAMMA 0.7, RL10 
50k)
  return pow(50e3 * pow(10, 0.7) / r, 1 / 0.7);
}

void readSlot(int i) {
  Slot& s = slots[i];
  float cm = (TEST_FORCE_FAULT_SLOT == i + 1) ? -1 : readDistanceCm(i);
  if (cm < 0) {                                           // invalid reading
    if (++s.invalid >= INVALID_LIMIT && !s.fault) {
      s.fault = true;
      Serial.printf("ALERT: sensor fault on %s\n", SLOT_ZONE[i]);
      pubFault(i, true);
      pubCounts();
    }
    return;
  }
  s.invalid = 0;
  s.cm = cm;
  if (s.fault) {                                          // recovery
    s.fault = false; s.init = false; s.agree = 0;
    pubFault(i, false);
  }
  bool cand = s.occupied;                                 // hysteresis
  if (cm < OCC_CM) cand = true;
  else if (cm > FREE_CM) cand = false;
  if (!s.init) {                                          // first valid reading: 
take it immediately
    s.init = true; s.occupied = cand; s.agree = 0;
    pubSlot(i); pubDistance(i); pubCounts();
    return;
  }
  if (cand != s.occupied) {                               // debounce
    if (++s.agree >= DEBOUNCE_N) {
      s.occupied = cand; s.agree = 0;
      pubSlot(i); pubDistance(i); pubCounts();
    }

  } else {
    s.agree = 0;
  }
  // live distance update while the slot state stays the same
  if (fabs(s.cm - s.pubCm) >= DIST_DELTA_CM && millis() - s.pubMs >= DIST_MIN_MS) 
pubDistance(i);
}

    else digitalWrite(PIN_LED[i], slots[i].init && slots[i].occupied);
  }
}
